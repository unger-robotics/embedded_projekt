% Warum Macro:
% Einheitliches Titel-Layout ohne Copy/Paste in der Dokumentstruktur.
% Cover-Bild füllt die Folie, Statuszeile am unteren Rand.
\newcommand{\titleslide}[2]{%
  \begin{frame}[plain,noframenumbering]
    \begin{tikzpicture}[remember picture,overlay]
      \node[at=(current page.center)] {%
        \includegraphics[width=\paperwidth,height=\paperheight]{#1}%
      };
      % Statuszeile ganz unten
      \node[anchor=south, text width=0.95\paperwidth, align=center] 
        at ([yshift=0.15cm]current page.south) {%
        \footnotesize\color{MutedText}#2%
      };
    \end{tikzpicture}
  \end{frame}%
}

% Video via Hyperlink (extern; bewusst kein Embedded-Video, weil Viewer/OS sonst zickt)
\newcommand{\videoslide}[4]{%
  \begin{frame}{#3}
    \framesubtitle{#4}
    \begin{center}
      \href{run:#1}{\includegraphics[width=12cm,height=6.75cm,keepaspectratio]{#2}}
      \vspace{0.3cm}

      \href{run:#1}{\textcolor{ArduinoTeal}{\textbf{Klick zum Abspielen}}}
    \end{center}
  \end{frame}%
}
